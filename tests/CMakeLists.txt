cmake_minimum_required(VERSION 3.12)
project(round_layout_test C CXX)
# Host test of the pinned LVGL with a dummy display and standard malloc.
add_compile_definitions(LV_CONF_SKIP=1 LV_MEM_CUSTOM=1)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../upstream/components/lvgl" lvgl)
add_executable(round_layout_test round_layout_test.c)
target_include_directories(round_layout_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../upstream/include")
target_link_libraries(round_layout_test PRIVATE lvgl)
enable_testing()
add_test(NAME round_layout COMMAND round_layout_test)
set_tests_properties(round_layout PROPERTIES TIMEOUT 20)
add_executable(round_demo_test round_demo_test.c ../upstream/main/gui/round_demo.c)
target_include_directories(round_demo_test PRIVATE ../upstream/include)
add_test(NAME round_demo COMMAND round_demo_test)
add_executable(round_saver_test round_saver_test.c ../upstream/main/gui/round_saver.c ../upstream/main/gui/round_demo.c ../upstream/main/gui/round_crew.c ../upstream/main/gui/round_expand.c)
target_include_directories(round_saver_test PRIVATE stubs ../upstream/include)
target_link_libraries(round_saver_test PRIVATE lvgl)
add_test(NAME round_saver COMMAND round_saver_test)
add_executable(peer_stub_test peer_stub_test.c ../upstream/main/core/esp_comm_manager.c)
target_include_directories(peer_stub_test PRIVATE stubs ../upstream/include)
add_test(NAME peer_stub COMMAND peer_stub_test)
set_tests_properties(round_demo round_saver peer_stub PROPERTIES TIMEOUT 20)
foreach(test_target round_demo_test round_saver_test peer_stub_test)
    target_compile_options(${test_target} PRIVATE -Wall -Wextra -Werror)
endforeach()
